# Function Program 17: Check Prime Number

def is_prime(n):
    if n < 2:
        return False
    for i in range(2, n):
        if n % i == 0:
            return False
    return True

print("Is prime:", is_prime(13))

# Function Program 18: Print Prime Numbers in a Range

def primes(start, end):
    for n in range(start, end + 1):
        if n > 1:
            prime = True
            for i in range(2, n):
                if n % i == 0:
                    prime = False
                    break
            if prime:
                print(n, end=" ")

primes(1, 30)

# Function Program 20: Reverse a Number

def reverse_number(n):
    reverse = 0
    while n > 0:
        digit = n % 10
        reverse = reverse * 10 + digit
        n = n // 10
    return reverse

print("Reversed number:", reverse_number(1234))

# Function Program 23: Check Palindrome Number

def is_palindrome(n):
    original = n
    reverse = 0
    while n > 0:
        reverse = reverse * 10 + n % 10
        n = n // 10
    return original == reverse

print("Is palindrome:", is_palindrome(1221))

# Function Program 24: Check Armstrong Number

def is_armstrong(n):
    original = n
    total = 0
    digits = len(str(n))
    while n > 0:
        digit = n % 10
        total = total + digit ** digits
        n = n // 10
    return total == original

print("Is Armstrong:", is_armstrong(153))

# Function Program 25: Greatest Common Divisor

def gcd(a, b):
    while b != 0:
        a, b = b, a % b
    return a

print("GCD:", gcd(24, 36))

# Function Program 26: Least Common Multiple

def lcm(a, b):
    x, y = a, b
    while y != 0:
        x, y = y, x % y
    return (a * b) // x

print("LCM:", lcm(12, 18))

# Function Program 42: Remove Duplicates from a List

def remove_duplicates(numbers):
    result = []
    for n in numbers:
        if n not in result:
            result.append(n)
    return result

nums = [1, 2, 2, 3, 4, 4, 5]
print("Without duplicates:", remove_duplicates(nums))
